SECURITY DEFINER-функции (точечное повышение) 
 
Цель: Реализовать безопасные операции через функции с повышенными 
привилегиями, сравнить методы контроля бизнес -правил. 
 
Были реализованы 3 функции для чувствительных операций в 
предметной области «Защита БД образовательных учреждений».  Код всех 3 
функций приведен в Приложении A. 
Первая функция (app.update_person_contacts()) позволяет роли 
app_writer изменять телефон и email в app.personal_files, не выдавая прямой 
доступ к таблице. Функция проверяет корректность идентификатора, 
телефона и email. 
Вторая функция (app.delete_grade()) используется для контролируемого 
удаления оценок. Прямой DELETE для app_writer запрещён, а функция 
дополнительно не позволяет удалять итоговые оценки типа final.  
Третья функция (app.enroll_student()) выполняет безопасную запись 
студента на проведение дисциплины. Проверяются существование и статус 
студента, соответствие учебной группе и отсутствие повторной записи.

-- app.update_person_contacts - функция для изменения телефона и электронной почты 
 
CREATE OR REPLACE FUNCTION app.update_person_contacts(p_id bigint, p_phone text, 
p_email text) 
RETURNS text 
LANGUAGE plpgsql 
SECURITY DEFINER 
SET search_path = pg_catalog 
AS $$ 
BEGIN 
 
if p_id is null or p_id <= 0 then 
    perform audit.log_function_call( 
    'app.update_person_contacts',session_user, 
    jsonb_build_object('person_id',p_id),false,'Неверный id'); 
    return 'Ошибка: неверный id'; 
end if; 
 
if p_phone is not null and p_phone !~ '^[0-9+-]{5,20}$' then 
    perform audit.log_function_call( 
    'app.update_person_contacts',session_user, 
    jsonb_build_object('person_id',p_id),false,'Неверный телефон'); 
    return 'Ошибка: неверный телефон'; 
end if; 
 
if p_email is not null and p_email not like '%@%.%' then 
    perform audit.log_function_call( 
    'app.update_person_contacts',session_user, 
    jsonb_build_object('person_id',p_id),false,'Неверный email'); 
    return 'Ошибка: неверный email'; 
end if; 
 
if not exists(select 1 from app.personal_files where person_id=p_id) then 
    perform audit.log_function_call( 
    'app.update_person_contacts',session_user, 
    jsonb_build_object('person_id',p_id),false,'Персональное дело не найдено'); 
    return 'Ошибка: персональное дело не найдено'; 
end if; 
 
update app.personal_files 
set phone=p_phone, email=p_email 
where person_id=p_id;

-- Создание тестовых таблиц для сравнения CHECK и триггера 
 
CREATE TABLE stg.semester_check_test ( 
    test_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY, 
    semester_number INTEGER NOT NULL, 
    start_date DATE NOT NULL, 
    end_date DATE NOT NULL, 
 
    CONSTRAINT chk_semester_rules 
    CHECK ( 
        semester_number IN (1, 2) 
        AND end_date > start_date 
        AND end_date - start_date <= 200 
    ) 
); 
 
CREATE TABLE stg.semester_trigger_test ( 
    test_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY, 
    semester_number INTEGER NOT NULL, 
    start_date DATE NOT NULL, 
    end_date DATE NOT NULL 
); 
 
 
-- Функция проверки бизнес-правил учебного семестра 
 
CREATE OR REPLACE FUNCTION stg.validate_semester_trigger() 
RETURNS trigger 
LANGUAGE plpgsql 
AS $$ 
BEGIN 
 
if NEW.semester_number <> 1 and NEW.semester_number <> 2 then 
    raise exception 'Номер семестра должен быть 1 или 2' 
    using errcode='23514'; 
end if; 
 
if NEW.end_date <= NEW.start_date then 
    raise exception 'Дата окончания должна быть позже даты начала' 
    using errcode='23514'; 
end if; 
 
if NEW.end_date - NEW.start_date > 200 then

24 
 
    raise exception 'Семестр не может длиться больше 200 дней' 
    using errcode='23514'; 
end if; 
 
return NEW; 
 
END; 
$$; 
 
 
-- Триггер проверки данных при добавлении и изменении учебного семестра 
 
CREATE TRIGGER trg_validate_semester 
BEFORE INSERT OR UPDATE 
ON stg.semester_trigger_test 
FOR EACH ROW 
EXECUTE FUNCTION stg.validate_semester_trigger();
